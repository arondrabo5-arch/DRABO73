async function loadMatches(){ const list = document.getElementById("matchesList"); list.textContent = "Chargement des rencontres..."; try { const response = await fetch( API_BASE + "/api/matches/upcoming" ); if (!response.ok) { throw new Error("Erreur API : " + response.status); } const data = await response.json(); matches = Array.isArray(data.matches) ? data.matches : []; document.getElementById("matchCount").textContent = matches.length; if (matches.length === 0) { list.textContent = "Aucun match à venir disponible."; return; } list.innerHTML = ""; matches.forEach((match, index) => { const item = document.createElement("div"); item.className = "match"; const title = document.createElement("h3"); title.textContent = (match.homeTeam?.name || "Domicile") + " - " + (match.awayTeam?.name || "Extérieur"); const competition = document.createElement("p"); competition.textContent = match.competition?.name || "Compétition non disponible"; const date = document.createElement("p"); date.textContent = match.utcDate ? new Date(match.utcDate).toLocaleString("fr-FR", { timeZone: "Africa/Ouagadougou", dateStyle: "short", timeStyle: "short" }) : "Date non disponible"; const status = document.createElement("p"); status.textContent = "Statut : " + match.status; const button = document.createElement("button"); button.className = "button"; button.textContent = "Analyser"; button.onclick = () => showAnalysis(index); item.append(title, competition, date, status, button); list.appendChild(item); }); } catch (error) { list.textContent = "Impossible de charger les matchs : " + error.message; } }
