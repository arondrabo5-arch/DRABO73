data.matches : []; count.textContent = String(matches.length); list.innerHTML = ""; if (matches.length === 0) { list.textContent = "Aucune rencontre à venir disponible actuellement."; return; } matches.forEach((match, index) => { const item = document.createElement("div"); item.className = "match"; const title = document.createElement("h3"); title.textContent = `${match.homeTeam?.name || "Domicile"} - ${match.awayTeam?.name || "Extérieur"}`; const competition = document.createElement("p"); competition.textContent = match.competition?.name || "Compétition non disponible"; const date = document.createElement("p"); date.textContent = match.utcDate ? new Date(match.utcDate).toLocaleString("fr-FR", { timeZone: "Africa/Ouagadougou", dateStyle: "short", timeStyle: "short" }) : "Date non disponible"; const status = document.createElement("p"); status.textContent = "Statut : " + (match.status || "Inconnu"); const button = document.createElement("button"); button.className = "button"; button.textContent = "Analyser"; button.addEventListener("click", () => { if (typeof showAnalysis === "function") { showAnalysis(index); } else { alert("La fonction d'analyse est introuvable."); } }); item.append( title, competition, date, status, button ); list.appendChild(item); }); } catch (error) { console.error("Erreur chargement matchs :", error); matches = []; count.textContent = "0"; list.textContent = "Impossible de charger les matchs : " + error.message; } }
